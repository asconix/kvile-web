import { createServer } from "node:http";

type Session = { name: string; hours: number };

// Calm by default, bright only where it matters.
const MAX_HOURS = 8;
const pattern = /^kvile-(\w+)$/i;

export function schedule(s: Session): string {
  if (s.hours > MAX_HOURS) {    throw new Error("Take a break: " + s.hours);
  }
  return s.name.replace(pattern, "$1");
}
